add_library(llm_engine
  src/device_info.cu
  src/loader.cpp
  kernels/vector_add.cu
)

target_include_directories(llm_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_link_libraries(llm_engine PUBLIC CUDA::cudart)
target_compile_definitions(llm_engine PUBLIC LLM_BUILD_SM=${LLM_CUDA_ARCH})
set_target_properties(llm_engine PROPERTIES
  CUDA_ARCHITECTURES ${LLM_CUDA_ARCH}
  CUDA_SEPARABLE_COMPILATION OFF
  POSITION_INDEPENDENT_CODE ON
)

add_library(llm::engine ALIAS llm_engine)

# One test executable per source file; no test framework by design (
# dependency policy) -- a test is a main() that returns non-zero on failure.
set(LLM_TESTS
  test_vector_add
  test_device_baseline
)

foreach(test_name IN LISTS LLM_TESTS)
  add_executable(${test_name} tests/${test_name}.cu)
  target_link_libraries(${test_name} PRIVATE llm_engine)
  set_target_properties(${test_name} PROPERTIES CUDA_ARCHITECTURES ${LLM_CUDA_ARCH})
  add_test(NAME ${test_name} COMMAND ${test_name})
  # A hung GPU test should fail the suite, not the CI job's wall clock.
  set_tests_properties(${test_name} PROPERTIES TIMEOUT 120)
endforeach()

# test_loader needs no GPU -- it's a pure host-side check of the offline
# binary format -- but it does need a fixture file generated by the same
# Python source (tools/convert/make_test_fixture.py) it's byte-compared
# against, regenerated at build time so it can never go stale.
set(LLM_FIXTURE_DIR ${CMAKE_CURRENT_BINARY_DIR}/test_fixtures)
set(LLM_FIXTURE_MODEL ${LLM_FIXTURE_DIR}/tiny_model.llmbin)
set(LLM_FIXTURE_EXPECTED_ROW ${LLM_FIXTURE_DIR}/expected_row.bin)

add_custom_command(
  OUTPUT ${LLM_FIXTURE_MODEL} ${LLM_FIXTURE_EXPECTED_ROW}
  COMMAND ${CMAKE_COMMAND} -E make_directory ${LLM_FIXTURE_DIR}
  COMMAND Python3::Interpreter ${CMAKE_CURRENT_SOURCE_DIR}/../tools/convert/make_test_fixture.py ${LLM_FIXTURE_DIR}
  DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/../tools/convert/make_test_fixture.py
          ${CMAKE_CURRENT_SOURCE_DIR}/../tools/convert/binformat.py
  COMMENT "Generating M1 loader test fixture from tools/convert/make_test_fixture.py"
)
add_custom_target(llm_loader_test_fixture DEPENDS ${LLM_FIXTURE_MODEL} ${LLM_FIXTURE_EXPECTED_ROW})

add_executable(test_loader tests/test_loader.cu)
target_link_libraries(test_loader PRIVATE llm_engine)
add_dependencies(test_loader llm_loader_test_fixture)
set_target_properties(test_loader PROPERTIES CUDA_ARCHITECTURES ${LLM_CUDA_ARCH})
add_test(NAME test_loader COMMAND test_loader ${LLM_FIXTURE_MODEL} ${LLM_FIXTURE_EXPECTED_ROW})
set_tests_properties(test_loader PROPERTIES TIMEOUT 120)
